# Pinned exactly, not loosely. This project's claims are specific numbers --
# 3/4 -> 0/4, zero false positives, 184 tests -- and verify_milestones.py
# asserts them against a live model. A dependency moving under us does not just
# risk a broken install, it risks the measured results no longer being the
# results these versions produced. LlamaIndex in particular ships breaking
# changes often.
#
# Aligned with the collaborator's fork: same llama-index-core, same Ollama
# embedding provider, so retrieval scores are comparable between the two
# repositories.
#
# requirements/lock.txt has the full transitive set for exact reproduction.
# requirements/huggingface.txt is optional, and only needed to reproduce the
# pre-Ollama retrieval scores still quoted in the docs.

python-dotenv==1.2.3

# PDF ingestion. llama-index-core alone has NO pdf reader: SimpleDirectoryReader
# falls back to reading the file as plain text, so a document loads as literal
# "%PDF-1.4 /BaseFont /Helvetica ..." instead of its prose. Nothing errors. A
# poisoned PDF would then look like it was defended against when in truth it was
# never read, which is the most dangerous way for this to fail.
llama-index-readers-file==0.7.0
